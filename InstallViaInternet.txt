۱.۱ — تنظیم hostname و hosts
bash

# روی master1:
hostnamectl set-hostname master1
# روی master2:  hostnamectl set-hostname master2
# روی master3:  hostnamectl set-hostname master3
# روی worker1:  hostnamectl set-hostname worker1

# روی همه نودها — /etc/hosts:
cat >> /etc/hosts <<EOF
10.0.0.10  master1
10.0.0.11  master2
10.0.0.12  master3
10.0.0.20  worker1
10.0.0.21  worker2
10.0.0.100 k8s-api
EOF

# ✅ چک:
hostname                                    # باید master1/worker1/... باشد
getent hosts master1 worker1 k8s-api        # باید IP برگرداند
ping -c 2 master2                           # باید resolve کند

۱.۲ — همگام‌سازی زمان (NTP)
bash

apt-get update && apt-get install -y chrony
systemctl enable --now chrony

# ✅ چک:
chronyc tracking | grep -E "Leap status|System time"
# انتظار: Leap status : Normal  +  System time : 0.000xxx seconds
timedatectl | grep "System clock synchronized: yes"

🔧 دلیل فنی: etcd حساس به clock skew است. اختلاف > 500ms باعث خطای lost leader یا request timeout می‌شود. kubelet هم با TLS cert not yet valid (اگر ساعت عقب باشد) fail می‌کند.
۱.۳ — غیرفعال کردن swap
bash

swapoff -a
sed -i '/\sswap\s/ s/^/#/' /etc/fstab

# ✅ چک:
swapon --show                  # باید خالی باشد
free -h | grep -i swap         # Swap: 0B 0B 0B

🔧 دلیل فنی: kubelet با failSwapOn: true (پیش‌فرض) اگر swap فعال باشد، کاملاً متوقف می‌شود. scheduler نمی‌تواند memory QoS را تضمین کند وقتی swap هست.

۱.۵ — پارامترهای Sysctl
bash

cat > /etc/sysctl.d/99-kubernetes-cri.conf <<EOF
# Networking
net.bridge.bridge-nf-call-iptables   = 1
net.bridge.bridge-nf-call-ip6tables  = 1
net.ipv4.ip_forward                  = 1
net.ipv4.conf.all.forwarding         = 1

# Conntrack (production: کلاستر با > 1000 pod)
net.netfilter.nf_conntrack_max       = 1048576
net.netfilter.nf_conntrack_tcp_timeout_established = 86400

# inotify (K8s از fsnotify برای watch استفاده می‌کند)
fs.inotify.max_user_watches         = 1048576
fs.inotify.max_user_instances       = 1024

# file descriptors
fs.file-max                         = 2097152

# ARP table (کلاستر بزرگ)
net.ipv4.neigh.default.gc_thresh1   = 4096
net.ipv4.neigh.default.gc_thresh2   = 8192
net.ipv4.neigh.default.gc_thresh3   = 16384

# performance
vm.swappiness                       = 0
vm.overcommit_memory                = 1
vm.panic_on_oom                     = 0
kernel.panic                        = 10
kernel.panic_on_oops                = 1
EOF

sysctl --system

# ✅ چک:
sysctl net.ipv4.ip_forward
# انتظار: net.ipv4.ip_forward = 1
sysctl net.bridge.bridge-nf-call-iptables
# انتظار: = 1
sysctl net.netfilter.nf_conntrack_max
# انتظار: = 1048576

🔧 دلیل فنی:

    ip_forward=1: بدون آن، ترافیک بین podها route نمی‌شود.

    bridge-nf-call-iptables=1: بدون آن، kube-proxy iptables روی bridge اعمال نمی‌شود → سرویس‌ها dead.

    nf_conntrack_max: 65536 پیش‌فرض برای کلاسترهای > 500 pod کافی نیست → nf_conntrack: table full, dropping packet در dmesg.

۲.۴ — نصب systemd unit برای containerd

[Unit]
Description=containerd container runtime
Documentation=https://containerd.io
After=network.target local-fs.target

[Service]
ExecStartPre=-/sbin/modprobe overlay
ExecStart=/usr/local/bin/containerd
Type=notify
Delegate=yes
KillMode=process
Restart=always
RestartSec=5
LimitNPROC=infinity
LimitCORE=infinity
LimitNOFILE=infinity
TasksMax=infinity
OOMScoreAdjust=-999

[Install]
WantedBy=multi-user.target
EOF

  gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg

# repository
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.31/deb/ /' | \
  tee /etc/apt/sources.list.d/kubernetes.list

apt-get install -y kubelet kubeadm kubectl
apt-mark hold kubelet kubeadm kubectl

# ✅ چک نسخه‌ها:

# انتظار: kubeadm version: &version.Info{Major:"1", Minor:"31", ...}

# انتظار: Kubernetes v1.31.x

# انتظار: Client Version: v1.31.x

    چرا apt-mark hold؟ جلوگیری از apt upgrade ناخواسته که kubelet را با API server ناسازگار می‌کند و کلاستر را خراب می‌کند.

۶.۴ — راه‌اندازی VIP با Keepalived (برای HA)

روی هر ۳ master:
bash

apt-get install -y keepalived

# master1 = MASTER, master2/3 = BACKUP
cat > /etc/keepalived/keepalived.conf <<EOF
vrrp_script check_apiserver {
  script "/etc/keepalived/check_apiserver.sh"
  interval 3
  weight -2
  fall 10
  rise 2
}

vrrp_instance VI_1 {
  state MASTER                     # روی master2/3 → BACKUP
  interface eth0                   # نام interface
  virtual_router_id 51
  priority 101                     # master1=101, master2=100, master3=99
  advert_int 1
  authentication {
    auth_type PASS
    auth_pass K8sHA_2024
  }
  virtual_ipaddress {
    10.0.0.100/24
  }
  track_script {
    check_apiserver
  }
}
EOF

cat > /etc/keepalived/check_apiserver.sh <<'EOF'
#!/bin/bash
errorExit() { echo "*** $*" 1>&2; exit 1; }
curl -sf --max-time 2 https://localhost:6443/healthz >/dev/null || errorExit "API down"
EOF
chmod +x /etc/keepalived/check_apiserver.sh

systemctl enable --now keepalived

# ✅ چک:
ip addr show eth0 | grep 10.0.0.100
# انتظار: روی master1 (MASTER) دیده می‌شود
curl -k https://10.0.0.100:6443/healthz
# انتظار: ok

🛡️ بخش ۹: Post-Install Production Hardening
۹.۱ — Backup etcd (روزانه)
bash

mkdir -p /backup/etcd

cat > /usr/local/bin/etcd-backup.sh <<'EOF'
#!/bin/bash
DATE=$(date +%Y%m%d-%H%M%S)
BACKUP=/backup/etcd/etcd-${DATE}.db
ETCDCTL_API=3 etcdctl snapshot save "$BACKUP" \
  --endpoints=https://127.0.0.1:2379 \
  --cacert=/etc/kubernetes/pki/etcd/ca.crt \
  --cert=/etc/kubernetes/pki/etcd/server.crt \
  --key=/etc/kubernetes/pki/etcd/server.key
ETCDCTL_API=3 etcdctl snapshot status "$BACKUP" -w table
# حذف بکاپ‌های > 7 روز
find /backup/etcd -name "*.db" -mtime +7 -delete
EOF
chmod +x /usr/local/bin/etcd-backup.sh

# cron:
cat > /etc/cron.d/etcd-backup <<EOF
0 2 * * * root /usr/local/bin/etcd-backup.sh >> /var/log/etcd-backup.log 2>&1
EOF

# ✅ تست دستی:
/usr/local/bin/etcd-backup.sh
ls -lh /backup/etcd/

۹.۲ — NetworkPolicy پیش‌فرض (default-deny)
bash

# ⚠️ قبل از اعمال، مطمئن شو NetworkPolicy ها را می‌فهمی
cat > /root/default-deny.yaml <<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: default-deny-ingress
  namespace: default
spec:
  podSelector: {}
  policyTypes: [Ingress]
---
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-dns-egress
  namespace: default
spec:
  podSelector: {}
  policyTypes: [Egress]
  egress:
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
      ports:
        - {protocol: UDP, port: 53}
        - {protocol: TCP, port: 53}
EOF

# فقط در namespaceهای مشخص اعمال کن — نه kube-system
