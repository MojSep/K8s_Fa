۳. بازنویسی check_host() — از host_path و label نود

def check_host():
    if not in_mode("node"):
        return
    # disk usage با prefix
    entries = []
    for path in DISK_PATHS:
        real = host_path(path)
        try:
            st = os.statvfs(real)
            used = 1 - (st.f_bavail / st.f_blocks)
            entries.append(({"path": path, "node": NODE_NAME}, used))
        except Exception:
            pass
    if entries:
        _replace_all_node(GAUGES["disk_usage"], entries)

    # swap
    try:
        with open(host_path("/proc/swaps")) as f:
            lines = [l for l in f.read().splitlines()[1:] if l.strip()]
        GAUGES["swap_active"].labels(node=NODE_NAME).set(1 if lines else 0)
    except Exception:
        GAUGES["swap_active"].labels(node=NODE_NAME).set(0)

    # sysctl — اگر hostPath mount /proc/sys باشد
    try:
        with open(host_path("/proc/sys/net/ipv4/ip_forward")) as f:
            ipf = f.read().strip() == "1"
        with open(host_path("/proc/sys/net/bridge/bridge-nf-call-iptables")) as f:
            bri = f.read().strip() == "1"
        GAUGES["sysctl_ok"].labels(node=NODE_NAME).set(1 if (ipf and bri) else 0)
    except Exception:
        GAUGES["sysctl_ok"].labels(node=NODE_NAME).set(0)

    # kernel modules
    try:
        with open(host_path("/proc/modules")) as f:
            mods = f.read()
        ok = all(f"{m} " in mods for m in ("overlay", "br_netfilter"))
        GAUGES["kernel_ok"].labels(node=NODE_NAME).set(1 if ok else 0)
    except Exception:
        GAUGES["kernel_ok"].labels(node=NODE_NAME).set(0)

    # kubelet/containerd — با hostPID، از /proc می‌خوانیم
    for svc in ("kubelet", "containerd"):
        pids = _proc_find(host_path("/proc"), svc)
        GAUGES[f"{svc}_active"].labels(node=NODE_NAME).set(1 if pids else 0)

def _proc_find(proc_root: str, name: str) -> list:
    """جستجوی process با name در /proc (hostPID لازم است)."""
    hits = []
    try:
        for pid in os.listdir(proc_root):
            if not pid.isdigit():
                continue
            try:
                with open(f"{proc_root}/{pid}/comm") as f:
                    if f.read().strip() == name:
                        hits.append(pid)
            except Exception:
                continue
    except Exception:
        pass
    return hits

def check_certs():
    """در cluster mode با kubeadm؛ در pod mode با openssl روی hostPath."""
    if not in_mode("cluster"):
        return
    import glob, re
    pki = host_path("/etc/kubernetes/pki")
    if not os.path.isdir(pki):
        log.debug("no pki dir at %s", pki); return

    entries = []
    for cert_file in glob.glob(f"{pki}/**/*.crt", recursive=True):
        try:
            r = subprocess.run(
                ["openssl", "x509", "-in", cert_file, "-noout", "-enddate"],
                capture_output=True, text=True, timeout=5)
            if r.returncode != 0:
                continue
            m = re.search(r"notAfter=(.+)", r.stdout)
            if not m:
                continue
            exp = datetime.strptime(m.group(1).strip(), "%b %d %H:%M:%S %Y %Z")
            exp = exp.replace(tzinfo=timezone.utc)
            days = int((exp - datetime.now(timezone.utc)).total_seconds() / 86400)
            name = os.path.relpath(cert_file, pki).replace("/", "_")
            entries.append(({"name": name}, days))
        except Exception:
            continue
    if entries:
        replace_all(GAUGES["cert_days"], entries)

عیب‌یابی رایج
مشکل	علت	رفع
cluster pod در Pending	روی master time-scheduling نشده	cluster.scheduleOnControlPlane: false + nodeSelector
kubectl not found در pod	image-build ناقص	docker build را از پوشه‌ی صحیح اجرا کن
permission denied روی /host	readOnlyRootFilesystem: true OK، اما hostPath mount اشتباه	readOnly: true روی volumeMount + type: Directory
hostPID fail	PodSecurity admission	namespace را با pod-security.kubernetes.io/enforce: privileged label کن، یا baseline معادل با runAsUser: 0
cert metric خالی	hostPath /etc/kubernetes/pki mount نشده	cluster.pkiHostPath را چک کن
kubelet_active = 0	hostPID غیرفعال	node.hostPID: true (پیش‌فرض است)
K8sHealthExporterDown	Prometheus نمی‌تواند scrape کند	NetworkPolicy، ServiceMonitor selector، namespace
helm test fail	image در registry نیست	docker login + push کن
drift بین cluster و node	label نود اشتباه	sum by (node)(k8s_health_disk_usage_ratio) را چک کن
نکته‌ی امنیتی

    DaemonSet با hostPID: true = دسترسی کامل به processهای نود. تنها در namespaceهای کنترل‌شده اجرا کن. readOnlyRootFilesystem: true + capabilities.drop: [ALL] + allowPrivilegeEscalation: false را حتماً داشته باش.

    ClusterRole read-only — هیچ permission نوشتنی نداشته باش به‌جز pods/exec که برای etcd check لازم است. اگر می‌خواهی این را هم حذف کنی، check_etcd را غیرفعال کن (در values: etcdCheck: false — این پارامتر را می‌توانی بعداً اضافه کنی).

    NetworkPolicy — پورت 9876/9877 را فقط از namespace مانیتورینگ باز بگذار:
    yaml

    networkPolicy:
      enabled: true
      prometheusNamespaceSelector:
        matchLabels: {kubernetes.io/metadata.name: monitoring}

🎯 جدول مقایسه‌ی دو مسیر
ویژگی	RBAC خام + Manifest	Helm Chart
سرعت راه‌اندازی	⚡ ۳۰ ثانیه	~۲ دقیقه
چند-کلاستر	دستی	--kube-context
نسخه‌بندی	دستی	helm history
Rollback	kubectl apply قبلی	helm rollback
Values override	sed	--set / -f prod.yaml
ServiceMonitor/Rule	دستی	خودکار (values toggle)
Alertmanager config	دستی	به‌صورت Secret تولید می‌شود
تست	دستی	helm test

توصیه: RBAC خام را برای تست dev استفاده کن، Helm Chart را در production با --kube-context روی همه کلاسترها.
🚀 Workflow نهایی (Production)
bash

# ۱. در CI — build و push
make build push IMAGE=ghcr.io/your-org/k8s-health TAG=$(git rev-parse --short HEAD)

# ۲. helm lint در PR
helm lint chart/k8s-health --strict

# ۳. تست در dev
helm upgrade --install k8s-health chart/k8s-health \
  --kube-context dev -n k8s-monitoring --create-namespace \
  -f values-dev.yaml

# ۴. roll-out prod (canary)
helm upgrade --install k8s-health chart/k8s-health \
  --kube-context prod -n k8s-monitoring --create-namespace \
  -f values-prod.yaml --wait --timeout 5m

# ۵. verify
helm test k8s-health --kube-context prod -n k8s-monitoring
kubectl --context prod -n k8s-monitoring get pods
curl -s http://prometheus/api/v1/query?query='up{job="k8s-health"}'
