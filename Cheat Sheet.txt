🎯 Cheat Sheet جامع DevOps — Kubernetes Production

    مرجع سریع روزانه برای مهندس DevOps در محیط عملیاتی. هر بخش با دستور، توضیح فنی، و نکته‌ی عملیاتی.

🔌 بخش ۱: Context & Namespace — قبل از هر کاری
bash

# ─── Context ───
kubectl config current-context              # کجا هستی؟
kubectl config get-contexts                 # همه contextها
kubectl config use-context prod-us          # تغییر
kubectl config view --minify --raw          # کامل
kubectl config view -o jsonpath='{.contexts[*].name}'

# ─── Namespace ───
kubectl config set-context --current --namespace=prod   # پیش‌فرض
kubectl get ns                              # لیست
kubectl ns prod                             # (kubens) سریع

# ─── Alias ضروری ───
alias k=kubectl
alias kg='kubectl get'
alias kd='kubectl describe'
alias kl='kubectl logs -f --tail=100'
alias kx='kubectl exec -it'
alias kaf='kubectl apply -f'
alias kdel='kubectl delete'
alias kctx='kubectl ctx'
alias kns='kubectl ns'
complete -F __start_kubectl k

🔧 نکته حیاتی: همیشه kubectl config current-context را قبل از delete یا drain چک کن. ۹۰٪ فاجعه‌های DevOps از context اشتباه است.
bash

# Prompt امن — نشان می‌دهد کجایی:
# در ~/.bashrc:
export PS1='\[\e[0;31m\]$(kubectl config current-context 2>/dev/null)\[\e[0m\] \w $ '

📦 بخش ۲: Pod Operations — ۸۰٪ کار روزانه
۲.۱ — مشاهده سریع
bash

kubectl get pods                              # default ns
kubectl get pods -A                           # همه ns
kubectl get pods -o wide                      # + IP + Node
kubectl get pods -o wide --sort-by=.spec.nodeName   # به ترتیب node
kubectl get pods -l app=api,tier=backend      # با label
kubectl get pods --field-selector=status.phase!=Running  # ناسالم
kubectl get pods -w                           # watch
kubectl get pods --show-labels                # labelها
kubectl get pods --show-kind --show-labels

۲.۲ — جزئیات
bash

kubectl describe pod <pod>                    # همیشه اول
kubectl describe pod <pod> | grep -A10 Events # فقط events
kubectl describe pod <pod> | grep -E "State|Reason|Exit|Message"
kubectl get pod <pod> -o yaml | kubectl neat  # yaml تمیز
kubectl get pod <pod> -o jsonpath='{.spec.nodeName}'
kubectl get pod <pod> -o jsonpath='{.status.podIP}'
kubectl get pod <pod> -o jsonpath='{.status.containerStatuses[*].restartCount}'

۲.۳ — لاگ
bash

kubectl logs <pod>                            # یک container
kubectl logs <pod> -c <container>             # container خاص
kubectl logs <pod> --previous                 # کانتینر کرش‌شده قبلی
kubectl logs <pod> -f --tail=100              # live
kubectl logs <pod> --since=10m                # ۱۰ دقیقه اخیر
kubectl logs <pod> --since-time="2024-11-08T10:00:00Z"
kubectl logs -l app=api --prefix -f           # چند pod
kubectl logs deploy/api --all-containers=true # همه containerها
kubectl logs job/backup                       # job
kubectl logs <pod> | grep -iE "error|exception|fail|panic"

# stern (بهتر):
stern -n prod deploy/api --since 10m
stern -n prod -l app=api --since 5m
stern -n prod 'api-.*' --since 10m
stern -n prod deploy/api -o json | jq 'select(.message | test("error";"i"))'

۲.۴ — Exec / Copy / Port-forward
bash

# ─── Exec ───
kubectl exec -it <pod> -- /bin/sh
kubectl exec -it <pod> -c sidecar -- bash
kubectl exec <pod> -- env                     # env vars
kubectl exec <pod> -- cat /etc/config.yaml
kubectl exec <pod> -- ps aux                  # processها
kubectl exec <pod> -- ss -tlnp                # listen ports
kubectl exec <pod> -- df -h                   # disk
kubectl exec <pod> -- nslookup api-svc        # DNS

# ─── Copy ───
kubectl cp <pod>:/path/file ./local
kubectl cp ./local <pod>:/path/file
kubectl cp <pod>:/path ./local -c sidecar

# ─── Port-forward ───
kubectl port-forward pod/<pod> 8080:80
kubectl port-forward svc/<svc> 8080:80
kubectl port-forward deploy/<deploy> 8080:80
kubectl port-forward -n monitoring svc/grafana 3000:3000 --address=0.0.0.0

۲.۵ — حذف
bash

kubectl delete pod <pod>
kubectl delete pod <pod> -n <ns> --grace-period=0 --force  # ⚠️ آخرین راه‌حل
kubectl delete pods -l app=api
kubectl delete pods --all -n dev --dry-run=client
kubectl delete pod <pod> --now                # فوری (v1.25+)

🔧 نکته: --force --grace-period=0 = SIGKILL مستقیم. اگر pod روی node کرش‌کرده باشد، ممکن است در apiserver گیر کند. ابتدا چک کن node واقعاً down است.

۳.۲ — Rollout
bash

kubectl rollout status deploy/<name>            # منتظر اتمام
kubectl rollout status deploy/<name> --timeout=5m
kubectl rollout history deploy/<name>           # تاریخچه

kubectl rollout undo deploy/<name>              # برگشت

kubectl rollout restart deploy/<name>           # restart بدون تغییر
kubectl rollout pause deploy/<name>             # متوقف (canary)
kubectl rollout resume deploy/<name>            # ادامه

🔧 نکته: rollout undo سریع‌ترین راه نجات از deploy بد است. اول این، بعد diagnose.

۷.۲ — Maintenance
bash

# Cordon (علامت‌گذاری)
kubectl cordon <node>

# Drain (تخلیه — با احترام به PDB)
kubectl drain <node> --ignore-daemonsets --delete-emptydir-data
kubectl drain <node> --ignore-daemonsets --delete-emptydir-data --force --timeout=300s

# Uncordon (برگشت)
kubectl uncordon <node>

# Taint
kubectl taint nodes <node> key=value:NoSchedule
kubectl taint nodes <node> key=value:NoExecute
kubectl taint nodes <node> key-                   # حذف

🔧 نکته حیاتی: Drain بدون --ignore-daemonsets اگر DaemonSet باشد خطا می‌دهد. بدون --delete-emptydir-data اگر emptyDir باشد خطا می‌دهد. همیشه هر دو flag.

۹.۲ — Resources Declared
bash

# Requests/Limits همه podها:
kubectl get pods -A -o json | jq -r '.items[] |
  .metadata.namespace + "/" + .metadata.name + " " +
  (.spec.containers[] | .name + ": req=" + (.resources.requests.cpu // "none") + "/" + (.resources.requests.memory // "none") + " lim=" + (.resources.limits.cpu // "none") + "/" + (.resources.limits.memory // "none"))'

# Podهای بدون limits:
kubectl get pods -A -o json | jq -r '.items[] |
  select(.spec.containers[].resources.limits == null) |
  "\(.metadata.namespace)/\(.metadata.name)"' | sort -u

# Podهای بدون requests:
kubectl get pods -A -o json | jq -r '.items[] |
  select(.spec.containers[].resources.requests == null) |
  "\(.metadata.namespace)/\(.metadata.name)"' | sort -u

۹.۳ — QoS
bash

kubectl get pods -A -o json | jq -r '.items[] |
  "\(.metadata.namespace)/\(.metadata.name)=\(.status.qosClass)"' | \
  sort -t= -k2 | grep -E "BestEffort|Burstable"

🔧 Eviction order: BestEffort → Burstable → Guaranteed. اگر pod حیاتی است، Guaranteed باشد (requests = limits).

🔍 بخش ۱۲: Top One-Liner های Production
۱۲.۱ — پیدا کردن مشکلات
bash

# همه podهای ناسالم:
kubectl get pods -A --no-headers | awk '$4!="Running" && $4!="Completed"'

# CrashLoopBackOff:
kubectl get pods -A | grep -E "CrashLoop|Error|ImagePull|ErrImage|Pending|Terminating"

# Restart بالا:
kubectl get pods -A --no-headers | awk '$5+0 > 5'

# Deployهای ناسالم:
kubectl get deploy -A --no-headers | awk '$3 != $4'

# PVC غیر Bound:
kubectl get pvc -A --no-headers | awk '$3 != "Bound"'

# Endpoint خالی:
kubectl get endpointslices -A -o json | jq -r '.items[] | select((.endpoints // []) | length == 0) | "\(.metadata.namespace)/\(.metadata.labels["kubernetes.io/service-name"])"'

# Nodeهای NotReady:
kubectl get nodes | grep -v " Ready " | grep -v "NAME"

# Node با pressure:
kubectl get nodes -o json | jq -r '.items[] | .metadata.name as $n | .status.conditions[] | select(.type|test("Pressure")) | select(.status=="True") | "\($n): \(.type)"'

# Podهای بدون owner (orphan):
kubectl get pods -A -o json | jq -r '.items[] | select(.metadata.ownerReferences == null and .metadata.annotations["kubernetes.io/config.mirror"] == null) | "\(.metadata.namespace)/\(.metadata.name)"'

# image هایی که از registry غیر داخلی هستند:
kubectl get pods -A -o jsonpath='{range .items[*]}{range .spec.containers[*]}{.image}{"\n"}{end}{end}' | \

۱۲.۳ — پیدا کردن منابع
bash

# کدام namespace بیشترین pod را دارد؟
kubectl get pods -A --no-headers | awk '{print $1}' | sort | uniq -c | sort -rn

# کدام node بیشترین pod را دارد؟
kubectl get pods -A -o wide --no-headers | awk '{print $8}' | sort | uniq -c | sort -rn

# Podهایی که به یک PVC وابسته‌اند:
kubectl get pods -A -o json | jq -r '.items[] | select(.spec.volumes[]?.persistentVolumeClaim) | "\(.metadata.namespace)/\(.metadata.name)"'

# کدام ConfigMap/Secret استفاده می‌شود؟
kubectl get pods -A -o json | jq -r '.items[] | .metadata.namespace as $ns | .metadata.name as $n | (.spec.volumes // [])[] | select(.configMap or .secret) | "\($ns)/\($n): " + ((.configMap.name // .secret.secretName))'

۱۲.۴ — Time-based
bash

# Podهای بالای ۲۴ ساعت:
kubectl get pods -A --no-headers | awk '{
  cmd="date -d \""$5"\" +%s"; cmd | getline created; close(cmd);
  age = systime() - created;
  if (age > 86400) print $1"/"$2" age="int(age/3600)"h"
}'

# رویدادهای ۵ دقیقه اخیر:
kubectl get events -A -o json | jq -r --arg t "$(date -u -d '5 minutes ago' +%Y-%m-%dT%H:%M:%SZ)" '
  .items[] | select(.lastTimestamp > $t) | "\(.lastTimestamp) \(.type) \(.involvedObject.name): \(.message)"'

۱۳.۷ — Emergency: etcd quorum lost

# ۱. روی master سالم:

# ۲. Restore از snapshot:

  --data-dir=/var/lib/etcd \

  --initial-cluster=<master>=https://<ip>:2380 \
  --initial-advertise-peer-urls=https://<ip>:2380

sleep 30

# ۴. چک:
kubectl get --raw='/readyz?verbose'

# ⚠️ کلاستر حالا تک‌نودی etcd است. masterهای دیگر را دوباره join کن.

🔧 بخش ۲۰: چک‌لیست سلامت روزانه
bash

#!/bin/bash
# daily-health.sh — اجرا در ابتدای هر شیفت

echo "═══ DAILY HEALTH $(date +%F) ═══"

# ── ۱. Nodes ──
not_ready=$(kubectl get nodes --no-headers | grep -vc " Ready ")
echo "Nodes NotReady: $not_ready $([ "$not_ready" -gt 0 ] && echo '❌')"

# ── ۲. Control Plane ──
cp_unhealthy=$(kubectl get pods -n kube-system --no-headers | grep -E "etcd|apiserver|controller|scheduler" | grep -vc "Running")
echo "Control Plane unhealthy: $cp_unhealthy $([ "$cp_unhealthy" -gt 0 ] && echo '❌')"

# ── ۳. All Pods ──
pods_unhealthy=$(kubectl get pods -A --no-headers | grep -vE "Running|Completed" | wc -l)
echo "Unhealthy Pods: $pods_unhealthy $([ "$pods_unhealthy" -gt 0 ] && echo '❌')"

# ── ۴. Deployments ──
deploy_unhealthy=$(kubectl get deploy -A --no-headers | awk '$3 != $4' | wc -l)
echo "Deployments unhealthy: $deploy_unhealthy $([ "$deploy_unhealthy" -gt 0 ] && echo '❌')"

# ── ۵. PVCs ──
pvc_unbound=$(kubectl get pvc -A --no-headers | awk '$3 != "Bound"' | wc -l)
echo "PVCs unbound: $pvc_unbound $([ "$pvc_unbound" -gt 0 ] && echo '❌')"

# ── ۶. Certificates ──
expiring=$(kubeadm certs check-expiration 2>/dev/null | grep -oE "[0-9]+d" | tr -d 'd' | awk '$1 < 30' | wc -l)
echo "Certs expiring <30d: $expiring $([ "$expiring" -gt 0 ] && echo '⚠️')"

# ── ۷. Disk ──
for p in / /var/lib/containerd /var/lib/etcd; do
  [ -d "$p" ] || continue
  usage=$(df "$p" 2>/dev/null | awk 'NR==2{print $5}' | tr -d '%')
  [ "$usage" -gt 85 ] && echo "Disk $p: ${usage}% ❌" || echo "Disk $p: ${usage}% ✅"
done

# ── ۸. API latency ──
t0=$(date +%s%N); kubectl get nodes >/dev/null 2>&1; t1=$(date +%s%N)
echo "API latency: $(( (t1-t0)/1000000 ))ms $([ $(( (t1-t0)/1000000 )) -gt 3000 ] && echo '❌')"

# ── ۹. etcd ──
kubectl -n kube-system exec etcd-$(hostname) -- etcdctl \
  --cacert=/etc/kubernetes/pki/etcd/ca.crt \
  --cert=/etc/kubernetes/pki/etcd/server.crt \
  --key=/etc/kubernetes/pki/etcd/server.key \
  endpoint health 2>&1 | grep -q "is healthy" && echo "etcd: ✅" || echo "etcd: ❌"

echo "═══ END ═══"
